-- ANTI-DETECTION CORE
-- Human-mimicking delay with variance jitter
-- ══════════════════════════════════════════
local function SmartWait(base, variance)
    variance = variance or base * 0.25
    local jitter = (math.random() - 0.5) * 2 * variance
    task.wait(math.max(0.04, base + jitter))
end

-- Rate-limited remote fire (max ~12/sec to stay under radar)
local _callCount  = 0
local _callWindow = os.clock()
local function SafeFire(remote, ...)
    if os.clock() - _callWindow >= 1 then
        _callCount  = 0
        _callWindow = os.clock()
    end
    _callCount += 1
    if _callCount > 12 then task.wait(0.08) end
    pcall(remote.FireServer, remote, ...)
end

local function SafeInvoke(remote, ...)
    if os.clock() - _callWindow >= 1 then
        _callCount  = 0
        _callWindow = os.clock()
    end
    _callCount += 1
    if _callCount > 12 then task.wait(0.08) end
    local ok, res = pcall(remote.InvokeServer, remote, ...)
    return ok and res or nil
end

-- Teleport with noise offset so position isn't pixel-perfect
local function SmartTP(hrp, targetCFrame)
    if not hrp then return end
    local noise = Vector3.new(
        (math.random() - 0.5) * 0.6,
        0,
        (math.random() - 0.5) * 0.6
    )
    hrp.CFrame = targetCFrame + noise
    SmartWait(0.45, 0.1)
end

-- Weather detection
local function GetCurrentWeather()
    local candidates = {
        workspace:FindFirstChild("Weather"),
        workspace:FindFirstChild("WeatherSystem"),
        workspace:FindFirstChild("GameManager"),
        ReplicatedStorage:FindFirstChild("WeatherData"),
    }
    for _, obj in ipairs(candidates) do
        if obj then
            local attr = obj:GetAttribute("CurrentWeather")
                or obj:GetAttribute("Weather")
                or obj:GetAttribute("ActiveWeather")
            if attr then return tostring(attr) end
            local child = obj:FindFirstChild("Current")
                or obj:FindFirstChild("ActiveWeather")
                or obj:FindFirstChild("WeatherType")
            if child and child.Value then return tostring(child.Value) end
        end
    end
    return "Any" -- fallback = always pass weather check
end

local function ToSet(t)
    if type(t) == "string" then return {[t]=true} end
    if type(t) == "table" then
        if t[1] then
            local s = {}
            for _, v in ipairs(t) do s[v] = true end
            return s
        end
        return t
    end
    return {}
end

-- Estimate value multiplier for ESP display
local function CalcMult(ripeness, variant, mutations)
    local rMult  = MUTATION_MULT[ripeness] or 1
    local vMult  = MUTATION_MULT[variant]  or 0
    local mSum   = 0
    if type(mutations) == "table" then
        for _, m in ipairs(mutations) do
            mSum += (MUTATION_MULT[m] or 0)
        end
    elseif type(mutations) == "string" and mutations ~= "None" then
        mSum = MUTATION_MULT[mutations] or 0
    end
    local mutTotal = math.max(1, mSum)
    return math.floor(rMult * (mutTotal + vMult) * 10) / 10
end

-- Get all mutations on a fruit (can be multi-mutation)
local function GetFruitMutations(fruit)
    local muts = {}
    local raw  = fruit:GetAttribute("Mutations") or fruit:GetAttribute("MutationList")
    if raw then
        if type(raw) == "string" then
            for m in raw:gmatch("[^,]+") do
                table.insert(muts, m:match("^%s*(.-)%s*$"))
            end
        end
    else
        local single = fruit:GetAttribute("Mutation") or "None"
        table.insert(muts, single)
    end
    return muts
end

-- BACKGROUND TASK MANAGER
-- Centralized start/stop with named coroutines
-- ══════════════════════════════════════════
local Tasks = {}

local function StartTask(name, fn, interval, variance)
    if Tasks[name] then return end
    Tasks[name] = true
    task.spawn(function()
        while Tasks[name] do
            pcall(fn)
            SmartWait(interval, variance or interval * 0.2)
        end
    end)
end

local function StopTask(name) Tasks[name] = nil end
local function IsRunning(name) return Tasks[name] == true end

-- ── AUTO HARVEST (weather + ripeness + mutation + variant filter) ──
local function DoAutoHarvest()
    local ripSet = ToSet(JinConfig.HarvestRipeness)
    local mutSet = ToSet(JinConfig.HarvestMutation)
    local varSet = ToSet(JinConfig.HarvestVariant)
    local wetSet = ToSet(JinConfig.HarvestWeather)

    if not (wetSet["Any"] or wetSet[GetCurrentWeather()]) then return end

    for _, plant in ipairs(clientPlants:GetChildren()) do
        local fruits = {}
        for _, child in ipairs(plant:GetChildren()) do
            if child.Name:lower():find("fruit") then
                table.insert(fruits, child)
            end
        end

        local processItem = function(item, uuid)
            local ripe = item:GetAttribute("RipenessStage") or "Unripe"
            local var  = item:GetAttribute("Variant") or "None"
            local muts = GetFruitMutations(item)
            local idx  = item:GetAttribute("GrowthAnchorIndex")

            if not (ripSet["Any"] or ripSet[ripe]) then return end
            if not (varSet["Any"] or varSet[var]) then return end

            -- Mutation check: pass if ANY selected mutation present
            local mutPass = mutSet["Any"]
            if not mutPass then
                for _, m in ipairs(muts) do
                    if mutSet[m] then mutPass = true break end
                end
            end
            if not mutPass then return end

            local payload = { Uuid = uuid }
            if idx then payload.GrowthAnchorIndex = idx end
            SafeFire(harvestEvent, { payload })
            SmartWait(0.06, 0.02)
        end

        if #fruits > 0 then
            local uuid = plant:GetAttribute("Uuid") or plant:GetAttribute("UUID")
            if uuid then
                for _, fruit in ipairs(fruits) do
                    processItem(fruit, uuid)
                end
            end
        else
            local uuid = plant:GetAttribute("Uuid") or plant:GetAttribute("UUID")
            if uuid then processItem(plant, uuid) end
        end
    end
end

-- ── SMART AUTO SHOVEL FRUIT ──
-- Uses 5 filter dimensions:
--   1. Ripeness (Unripe / Ripened / Lush)
--   2. Variant  (None / Silver / Gold)
--   3. Mutations (any selection)
--   4. Min weight to KEEP (kg)
--   5. Max weight to SHOVEL (fruits too big get kept)
-- Mode:
--   "Remove Junk"    – shovel fruits that DON'T match keep-filters
--   "Remove All"     – shovel every fruit unconditionally
--   "Keep Valuable"  – shovel everything EXCEPT Gold/Lush/high-mutation

local function FruitPassesShovelKeep(fruit)
    local mode   = JinConfig.ShovelMode or "Remove Junk"
    local weight = tonumber(fruit:GetAttribute("Weight") or fruit:GetAttribute("Kg") or fruit:GetAttribute("Size")) or 0
    local minKg  = tonumber(JinConfig.ShovelMinKg) or 0
    local maxKg  = tonumber(JinConfig.ShovelMaxKg) or 999

    if mode == "Remove All" then
        return false -- shovel everything, nothing passes keep
    end

    local ripe = fruit:GetAttribute("RipenessStage") or "Unripe"
    local var  = fruit:GetAttribute("Variant") or "None"
    local muts = GetFruitMutations(fruit)

    local keepRipeSet  = ToSet(JinConfig.ShovelKeepRipeness)
    local keepVarSet   = ToSet(JinConfig.ShovelKeepVariant)
    local keepMutSet   = ToSet(JinConfig.ShovelKeepMutation)

    -- Weight gate: if below minKg → always shovel
    if weight > 0 and weight < minKg then return false end
    -- Weight gate: if above maxKg → always keep (too valuable to delete)
    if maxKg > 0 and weight > maxKg then return true end

    if mode == "Keep Valuable" then
        -- Keep if Gold or Lush or has a mutation with mult > 3x
        if var == "Gold" then return true end
        if ripe == "Lush" then return true end
        for _, m in ipairs(muts) do
            if (MUTATION_MULT[m] or 0) >= 3.0 then return true end
        end
        return false -- everything else gets shoveled
    end

    -- "Remove Junk" mode: keep if passes ALL three filter checks
    local ripePass = keepRipeSet["Any"] or keepRipeSet[ripe]
    local varPass  = keepVarSet["Any"] or keepVarSet[var]
    local mutPass  = keepMutSet["Any"]
    if not mutPass then
        for _, m in ipairs(muts) do
            if keepMutSet[m] then mutPass = true break end
        end
        if not mutPass and muts[1] == "None" and keepMutSet["None"] then
            mutPass = true
        end
    end

    return ripePass and varPass and mutPass
end

local function DoAutoShovelFruit()
    if not shovelEvent then return end
    for _, plant in ipairs(clientPlants:GetChildren()) do
        local uuid = plant:GetAttribute("Uuid") or plant:GetAttribute("UUID")
        if not uuid then continue end

        -- Multi-fruit plants
        local fruits = {}
        for _, child in ipairs(plant:GetChildren()) do
            if child.Name:lower():find("fruit") then
                table.insert(fruits, child)
            end
        end

        if #fruits > 0 then
            for _, fruit in ipairs(fruits) do
                local keep = FruitPassesShovelKeep(fruit)
                if not keep then
                    local idx = fruit:GetAttribute("GrowthAnchorIndex")
                    local payload = { Uuid = uuid, Type = "Fruit" }
                    if idx then payload.GrowthAnchorIndex = idx end
                    SafeInvoke(shovelEvent, payload)
                    SmartWait(0.09, 0.03)
                end
            end
        else
            -- Single-fruit plant
            local keep = FruitPassesShovelKeep(plant)
            if not keep then
                SafeInvoke(shovelEvent, { Uuid = uuid, Type = "Fruit" })
                SmartWait(0.09, 0.03)
            end
        end
    end
end

-- ── AUTO SHOVEL TREE (dead / finished only) ──
local function DoAutoShovelTree()
    if not shovelEvent then return end
    for _, plant in ipairs(clientPlants:GetChildren()) do
        local uuid = plant:GetAttribute("Uuid") or plant:GetAttribute("UUID")
        local dead = plant:GetAttribute("IsDead") or plant:GetAttribute("Dead")
            or plant:GetAttribute("Finished") or plant:GetAttribute("FullyGrown")
        if uuid and dead then
            SafeInvoke(shovelEvent, { Uuid = uuid, Type = "Tree" })
            SmartWait(0.1, 0.03)
        end
    end
end

    end
end

-- ── AUTO PLANT ──
local function DoAutoPlant()
    local seedSet   = ToSet(JinConfig.SeedSelect)
    local seeds     = {}
    for k, v in pairs(seedSet) do if v then table.insert(seeds, k) end end
    if #seeds == 0 then return end

    local seedName  = seeds[math.random(1, #seeds)]
    local char      = LocalPlayer.Character
    local human     = char and char:FindFirstChild("Humanoid")
    local backpack  = LocalPlayer:FindFirstChild("Backpack")
    local keyword   = seedName .. " Seed"
    local tool      = nil

    if char then
        for _, c in ipairs(char:GetChildren()) do
            if c:IsA("Tool") and c.Name:find(keyword) then tool = c break end
        end
    end
    if not tool and backpack and human then
        for _, c in ipairs(backpack:GetChildren()) do
            if c:IsA("Tool") and c.Name:find(keyword) then
                human:EquipTool(c)
                SmartWait(0.2, 0.05)
                tool = c
                break
            end
        end
    end
    if not tool then return end

    local pos
    local mode = JinConfig.PlantMode
    if mode == "Player Position" then
        if char and char:FindFirstChild("HumanoidRootPart") then
            pos = char.HumanoidRootPart.Position
        end
    elseif mode == "Custom Position" then
        pos = Vector3.new(
            tonumber(JinConfig.CustomPlantX) or 0,
            0,
            tonumber(JinConfig.CustomPlantZ) or 0
        )
    else -- Random
        local myPlot = GetMyPlot()
        if myPlot and myPlot:FindFirstChild("PlantableArea") then
            local parts = {}
            for _, p in ipairs(myPlot.PlantableArea:GetChildren()) do
                if p:IsA("BasePart") then table.insert(parts, p) end
            end
            if #parts > 0 then
                pos = GetRandPosInPart(parts[math.random(1, #parts)])
            end
        end
    end

    if pos then SafeInvoke(plantEvent, seedName, pos) end
end

-- ── AUTO SELL ──
local function DoAutoSell(force)
    local hrp = GetHRP()
    if not hrp then return end
    if not force then
        local thresh = tonumber(JinConfig.BackpackFullThresh) or 35
        if JinConfig.ForceSellOnFull and GetBackpackCount() >= thresh then force = true end
    end
    TeleportToShop("Sell Stand")
    SafeInvoke(sellEvent, "SellAll")
    SmartWait(0.4, 0.1)
end

-- ── AUTO BUY SEEDS (best affordable or selected) ──
local function DoAutoBuySeeds()
    local shillings = GetShillings()
    if not shillings then return end
    local didTP = false

    if JinConfig.AutoBuyBestSeed or JinConfig.BuyMode == "Buy Best Seed" then
        for i = #seedShopData, 1, -1 do
            local d   = seedShopData[i]
            local key = d.Key
            if shillings.Value >= d.Price and GetSeedStock(key) > 0 then
                if not didTP then didTP = TeleportToShop("Seed Shop") end
                local bought = 0
                while JinConfig.AutoBuySeeds and shillings.Value >= d.Price
                    and GetSeedStock(key) > 0 and bought < 10 do
                    SafeInvoke(purchaseEvent, "SeedShop", d.Name)
                    SmartWait(0.13, 0.04)
                    bought += 1
                end
                break
            end
        end
    else
        local sel = ToSet(JinConfig.TargetBuySeed)
        for sName, isOn in pairs(sel) do
            if not isOn then continue end
            local price, key = 0, ""
            for _, d in ipairs(seedShopData) do
                if d.Name == sName then price = d.Price key = d.Key break end
            end
            if shillings.Value >= price and GetSeedStock(key) > 0 then
                if not didTP then didTP = TeleportToShop("Seed Shop") end
                local b = 0
                while JinConfig.AutoBuySeeds and shillings.Value >= price
                    and GetSeedStock(key) > 0 and b < 10 do
                    SafeInvoke(purchaseEvent, "SeedShop", sName)
                    SmartWait(0.13, 0.04)
                    b += 1
                end
            end
            if not JinConfig.AutoBuySeeds then break end
        end
    end
end

-- ── AUTO BUY GEARS ──
local function DoAutoBuyGears()
    local shillings = GetShillings()
    if not shillings then return end
    local didTP = false

    if JinConfig.AutoBuyBestGear then
        for i = #gearShopData, 1, -1 do
            local d = gearShopData[i]
            if shillings.Value >= d.Price and GetGearStock(d.Name) > 0 then
                if not didTP then didTP = TeleportToShop("Gear Shop") end
                SafeInvoke(purchaseEvent, "GearShop", d.Name)
                SmartWait(0.2, 0.05)
                break
            end
        end
    else
        local sel = ToSet(JinConfig.TargetBuyGear)
        for gName, isOn in pairs(sel) do
            if not isOn then continue end
            local price = 0
            for _, d in ipairs(gearShopData) do
                if d.Name == gName then price = d.Price break end
            end
            if shillings.Value >= price and GetGearStock(gName) > 0 then
                if not didTP then didTP = TeleportToShop("Gear Shop") end
                local b = 0
                while JinConfig.AutoBuyGears and shillings.Value >= price
                    and GetGearStock(gName) > 0 and b < 5 do
                    SafeInvoke(purchaseEvent, "GearShop", gName)
                    SmartWait(0.15, 0.04)
                    b += 1
                end
            end
            if not JinConfig.AutoBuyGears then break end
        end
    end
end

local RIPE_COLORS = {
    Lush    = Color3.fromRGB(0,   255, 80),
    Ripened = Color3.fromRGB(255, 200, 0),
    Unripe  = Color3.fromRGB(160, 160, 160),
}
local VAR_COLORS = {
    Gold   = Color3.fromRGB(255, 215, 0),
    Silver = Color3.fromRGB(200, 200, 255),
    None   = Color3.fromRGB(200, 200, 200),
}

local function DoPlantESP()
    ClearESP()
    if not JinConfig.PlantESP then return end

    for _, plant in ipairs(clientPlants:GetChildren()) do
        local root = plant:FindFirstChild("HumanoidRootPart")
            or plant:FindFirstChildWhichIsA("BasePart")
        if not root then continue end

        local ripe  = plant:GetAttribute("RipenessStage") or "Unripe"
        local var   = plant:GetAttribute("Variant") or "None"
        local muts  = GetFruitMutations(plant)
        local weight = plant:GetAttribute("Weight") or plant:GetAttribute("Kg") or "?"
        local mult  = JinConfig.ESPShowMult and CalcMult(ripe, var, muts) or nil

        local mutStr = table.concat(muts, "+")
        local wStr   = type(weight) == "number" and string.format("%.2fkg", weight) or tostring(weight)
        local mStr   = mult and (" | ×" .. mult) or ""

        local bb = Instance.new("BillboardGui")
        bb.Size          = UDim2.new(0, 140, 0, 55)
        bb.StudsOffset   = Vector3.new(0, 3.5, 0)
        bb.AlwaysOnTop   = true
        bb.Adornee       = root
        bb.Parent        = ESPFolder

        local bg = Instance.new("Frame")
        bg.Size                  = UDim2.new(1,0,1,0)
        bg.BackgroundColor3      = Color3.fromRGB(0,0,0)
        bg.BackgroundTransparency= 0.4
        bg.BorderSizePixel       = 0
        bg.Parent                = bb

        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(0, 4)
        corner.Parent       = bg

        local label = Instance.new("TextLabel")
        label.Size                   = UDim2.new(1,-4,1,-4)
        label.Position               = UDim2.new(0,2,0,2)
        label.BackgroundTransparency = 1
        label.TextColor3             = RIPE_COLORS[ripe] or Color3.fromRGB(255,255,255)
        label.Font                   = Enum.Font.GothamBold
        label.TextScaled             = true
        label.Text                   = string.format(
            "[%s] %s\n%s | %s%s",
            ripe, var, mutStr, wStr, mStr
        )
        label.Parent = bg

        -- Gold/Silver name color
        if var ~= "None" then
            label.TextColor3 = VAR_COLORS[var]
        end
    end
end

-- TAB 7: QUEST & CODES
-- ══════════════════════════════════════════
local QuestTab = Window:Tab({ Title = "Quest", Icon = "scroll-text" }) do

    QuestTab:Section({ Title = "Full Auto Quest" })
    QuestTab:Toggle({
        Title = "Enable Full Auto Quest",
        Desc  = "Loops: Buy seeds → Plant → Harvest → Sell to complete quests",
        Value = JinConfig.AutoQuest,
        Callback = function(v)
            JinConfig.AutoQuest = v CS()
            if v then StartTask("quest", DoAutoQuest, 2)
            else StopTask("quest") end
        end
    })

    QuestTab:Section({ Title = "Auto Redeem Codes" })
    QuestTab:Textbox({
        Title = "Add Code (press Enter)",
        Placeholder = "FREEGOLD",
        ClearTextOnFocus = true,
        Callback = function(v)
            if v and v ~= "" then
                JinConfig.RedeemCodes = JinConfig.RedeemCodes or {}
                table.insert(JinConfig.RedeemCodes, v)
                CS()
                Window:Notify({ Title = "Code Added", Desc = v, Time = 3 })
            end
        end
    })
    QuestTab:Button({
        Title = "Redeem All Codes Now",
        Callback = function() task.spawn(DoAutoRedeem) end
    })
    QuestTab:Button({
        Title = "Clear Saved Codes",
        Callback = function()
            JinConfig.RedeemCodes = {}
            CS()
            Window:Notify({ Title = "Codes Cleared", Time = 2 })
        end
    })
    QuestTab:Toggle({
        Title = "Auto Redeem on Load",
        Value = JinConfig.AutoRedeem,
        Callback = function(v) JinConfig.AutoRedeem = v CS() end
    })

end

local SettingsTab = Window:Tab({ Title = "Settings", Icon = "wrench" }) do

    SettingsTab:Section({ Title = "Info" })
    SettingsTab:Button({
        Title = "Copy Discord Link",
        Desc  = "discord.gg/k2wdyy8QMN",
        Callback = function()
            if setclipboard then setclipboard("https://discord.gg/k2wdyy8QMN") end
            Window:Notify({ Title = "Copied!", Desc = "discord.gg/k2wdyy8QMN", Time = 3 })
        end
    })

    SettingsTab:Section({ Title = "Anti-AFK" })
    local AntiAfkConn
    SettingsTab:Toggle({
        Title = "Anti-AFK",
        Desc  = "Prevents idle kick",
        Value = JinConfig.AntiAFK,
        Callback = function(v)
            JinConfig.AntiAFK = v CS()
            if v then
                AntiAfkConn = LocalPlayer.Idled:Connect(function()
                    VirtualUser:Button2Down(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
                    task.wait(1)
                    VirtualUser:Button2Up(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
                end)
            else
                if AntiAfkConn then AntiAfkConn:Disconnect() AntiAfkConn = nil end
            end
        end
    })

    -- Auto-connect if enabled
    if JinConfig.AntiAFK then
        AntiAfkConn = LocalPlayer.Idled:Connect(function()
            VirtualUser:Button2Down(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
            task.wait(1)
            VirtualUser:Button2Up(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
        end)
    end

    SettingsTab:Section({ Title = "Config" })
    SettingsTab:Toggle({
        Title = "Auto Save", Desc = "Saves on every change",
        Value = JinConfig.AutoSave,
        Callback = function(v)
            JinConfig.AutoSave = v SaveConfig()
            Window:Notify({ Title = "Auto Save " .. (v and "ON" or "OFF"), Time = 2 })
        end
    })
    SettingsTab:Toggle({
        Title = "Auto Load", Desc = "Loads config on inject",
        Value = JinConfig.AutoLoad,
        Callback = function(v)
            JinConfig.AutoLoad = v SaveConfig()
            Window:Notify({ Title = "Auto Load " .. (v and "ON" or "OFF"), Time = 2 })
        end
    })
    SettingsTab:Button({
        Title = "Save Config Now",
        Callback = function()
            SaveConfig()
            Window:Notify({ Title = "Saved!", Time = 2 })
        end
    })
    SettingsTab:Button({
        Title = "Reset to Defaults",
        Callback = function()
            JinConfig = DefaultConfig
            SaveConfig()
            Window:Notify({ Title = "Reset Done", Desc = "Rejoin for full effect", Time = 4 })
        end
    })

end
